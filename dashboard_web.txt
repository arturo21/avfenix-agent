            try {
                const res = await fetch(`${apiBase}/api/health`);

                    connIndicator.className = "w-3.5 h-3.5 rounded-full bg-green-500 shadow-lg shadow-green-500/30";
                    connText.innerText = "Conectado";

        async function loadDocuments() {

                        return;
                    }

                    documentList.innerHTML = '';
                    docs.forEach(doc => {
                        const isUrl = doc.startsWith("WEB_");
                        const icon = isUrl ? '🌐' : '📄';

                        div.className = "flex items-center justify-between p-2.5 bg-gray-50 border border-gray-100 rounded-lg hover:bg-gray-100 transition";

                            </div>
                            <button onclick="deleteDocument('${doc}')" class="p-1 hover:bg-red-50 text-gray-400 hover:text-red-500 rounded transition">
                                <svg class="w-4 h-4" fill="none" stroke="currentColor" viewBox="0 0 24 24"><path stroke-linecap="round" stroke-linejoin="round" stroke-width="2" d="M19 7l-.867 12.142A2 2 0 0116.138 21H7.862a2 2 0 01-1.995-1.858L5 7m5 4v6m4-6v6m1-10V4a1 1 0 00-1-1h-4a1 1 0 00-1 1v3M4 7h16"></path></svg>

        async function loadAnalytics() {
            const apiBase = apiUrlInput.value.trim();
            try {
                const res = await fetch(`${apiBase}/api/analytics`);
                if (!res.ok) return;

                const data = await res.json();
                
                document.getElementById('stat-total-users').innerText = data.total_users || 0;
                document.getElementById('stat-total-messages').innerText = data.total_messages || 0;
                document.getElementById('stat-total-leads').innerText = data.total_leads || 0;

                const msgChannels = data.messages_by_channel || {};
                const totalMsgs = Math.max(data.total_messages || 1, 1);

                const channels = ['web', 'whatsapp', 'instagram', 'facebook'];
                channels.forEach(ch => {
                    const count = msgChannels[ch] || 0;
                    const pct = Math.round((count / totalMsgs) * 100);
                    
                    const cntElem = document.getElementById(`cnt-${ch}`);
                    const barElem = document.getElementById(`bar-${ch}`);
                    
                    if (cntElem) cntElem.innerText = `${count} msgs (${pct}%)`;
                    if (barElem) barElem.style.width = `${pct}%`;
                });

                const leadsTable = document.getElementById('leads-table-body');
                const leads = data.recent_leads || [];

                if (leads.length === 0) {
                    leadsTable.innerHTML = `
                        <tr>
                            <td colspan="6" class="p-8 text-center text-gray-400">

                    if (lead.channel === 'instagram') badgeColor = "bg-pink-100 text-pink-800";

                    tr.innerHTML = `
                        <td class="p-3 font-mono text-gray-400">#${lead.id}</td>

                    loadDocuments();
                    showNotification(`Se eliminó "${filename}".`, 'success');
                } else {
                    showNotification("No se pudo eliminar la fuente.", "error");
                }
            } catch (err) {
                showNotification("Error de red.", "error");
            }
        }

        scrapeForm.addEventListener('submit', async (e) => {
            e.preventDefault();
            const url = urlInput.value.trim();
            if (!url) return;

            const apiBase = apiUrlInput.value.trim();

            btnScrape.innerText = "Scrapeando...";

            try {
                const res = await fetch(`${apiBase}/api/scrape_url`, {
                    method: 'POST',

                    const errData = await res.json();
                    showNotification(errData.error || "Error al scrapear la URL.", 'error');

            } catch (err) {
                showNotification("Error de conexión al scrapear la URL.", 'error');

        async function uploadFile(file) {
            const apiBase = apiUrlInput.value.trim();
            const formData = new FormData();
            formData.append('file', file);

            progressContainer.classList.remove('hidden');
            uploadFeedback.classList.add('hidden');
            progressBar.style.width = '30%';
            progressPercent.innerText = '30%';

            try {
                const res = await fetch(`${apiBase}/api/upload`, { method: 'POST', body: formData });
                progressBar.style.width = '100%';
                progressPercent.innerText = '100%';

                if (res.ok) {
                    const data = await res.json();
                    showNotification(`¡Éxito! "${data.filename}" indexado con ${data.chunks_count} fragmentos.`, 'success');
                    loadDocuments();
                } else {
                    const errData = await res.json();
                    showNotification(errData.error || "Error al indexar.", 'error');
                }
            } catch (err) {
                showNotification("No hay comunicación con el servidor.", 'error');
            } finally {
                setTimeout(() => { progressContainer.classList.add('hidden'); progressBar.style.width = '0%'; }, 3000);
            }
        }

        async function sendChatMessage(text) {
            appendMessage('user', text);
            chatInput.value = '';

            const botMsgId = appendMessage('assistant', `
                <div class="flex items-center space-x-2 py-1">
                    <span class="w-2 h-2 rounded-full bg-primary-600 animate-bounce"></span>
                    <span class="w-2 h-2 rounded-full bg-primary-600 animate-bounce [animation-delay:0.2s]"></span>
                    <span class="w-2 h-2 rounded-full bg-primary-600 animate-bounce [animation-delay:0.4s]"></span>
                </div>
            `, true);

            const apiBase = apiUrlInput.value.trim();
            try {
                const res = await fetch(`${apiBase}/api/chat`, {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify({ message: text, output_mode: chatMode.value })
                });

                if (res.ok) {
                    const data = await res.json();
                    updateBotMessage(botMsgId, data.response, data.sources || [], data.audio_url, data.suggestions || []);
                } else {
                    const errData = await res.json();
                    updateBotMessage(botMsgId, `Error en el servidor: ${errData.error || 'Respuesta fallida'}`);
                }
            } catch (err) {
                updateBotMessage(botMsgId, `Error de conexión con la API en ${apiBase}`);
            }
        }

        window.sendSuggestion = function(text) {
            sendChatMessage(text);
        };

        function appendMessage(role, text, isRawHtml = false) {
            const msgId = 'msg-' + Date.now() + Math.random().toString(36).substr(2, 4);
            const div = document.createElement('div');
            
            if (role === 'user') {
                div.className = "flex items-start justify-end space-x-3";
                div.innerHTML = `
                    <div class="bg-primary-600 text-white rounded-2xl rounded-tr-none p-3.5 shadow-sm max-w-lg">
                        <p class="text-sm">${text}</p>
                    </div>
                    <div class="w-8 h-8 rounded-full bg-primary-800 flex items-center justify-center text-white text-xs font-bold flex-shrink-0">Tú</div>
                `;
            } else {
                div.className = "flex items-start space-x-3 max-w-xl";
                div.innerHTML = `
                    <div class="w-8 h-8 rounded-full bg-primary-600 flex-shrink-0 flex items-center justify-center text-white text-xs font-bold">AV</div>
                    <div class="bg-white border border-gray-100 rounded-2xl rounded-tl-none p-3.5 shadow-sm flex-1" id="${msgId}-body">
                        <div class="text-sm text-gray-800">${isRawHtml ? text : formatText(text)}</div>
                    </div>
                `;
            }

            chatBox.appendChild(div);
            chatBox.scrollTop = chatBox.scrollHeight;
            return msgId;
        }

        function updateBotMessage(msgId, text, sources = [], audioUrl = null, suggestions = []) {
            const bodyDiv = document.getElementById(`${msgId}-body`);
            if (!bodyDiv) return;

            let htmlContent = `<p class="text-sm text-gray-800">${formatText(text)}</p>`;

            if (audioUrl) {
                const fullAudio = `${apiUrlInput.value.trim()}${audioUrl}`;
                htmlContent += `
                    <div class="mt-3 bg-gray-50 p-2 rounded-lg border border-gray-100 flex items-center justify-between">
                        <span class="text-xs text-gray-500 font-semibold uppercase tracking-wide">Voz Generada (TTS)</span>
                        <audio controls class="h-8 max-w-[200px]" src="${fullAudio}"></audio>
                    </div>
                `;
            }

            if (suggestions && suggestions.length > 0) {
                let sugButtons = '';
                suggestions.forEach(sug => {
                    const safeSug = sug.replace(/"/g, '&quot;').replace(/'/g, "\\'");
                    sugButtons += `
                        <button onclick="sendSuggestion('${safeSug}')" class="px-3 py-1.5 bg-sky-50 hover:bg-primary-600 hover:text-white text-primary-700 text-xs font-medium rounded-full border border-sky-200 transition text-left shadow-sm">
                            💡 ${sug}
                        </button>
                    `;
                });

                htmlContent += `
                    <div class="mt-3 border-t border-gray-100 pt-2.5">
                        <span class="text-[11px] font-bold text-gray-500 block mb-1.5">Sugerencias para avanzar:</span>
                        <div class="flex flex-wrap gap-1.5">
                            ${sugButtons}
                        </div>
                    </div>
                `;
            }

            if (sources.length > 0) {
                let sourceItems = '';
                sources.forEach(src => {
                    sourceItems += `
                        <div class="p-2 bg-gray-50 rounded border border-gray-100 text-xs">
                            <div class="flex justify-between font-bold text-gray-700 mb-0.5">
                                <span class="truncate max-w-[250px]">📌 ${src.filename}</span>
                                <span class="text-[10px] text-primary-600">Relevancia: ${src.score}</span>
                            </div>
                            <div class="text-[10px] text-gray-500">Sección / Pág: ${src.page_number}</div>
                            <p class="text-[10px] text-gray-400 italic mt-1 line-clamp-2">"${src.text}"</p>
                        </div>
                    `;
                });

                htmlContent += `
                    <div class="mt-3 border-t border-gray-100 pt-2">
                        <details class="group">
                            <summary class="list-none flex items-center justify-between text-xs text-primary-600 font-semibold cursor-pointer select-none">
                                <span>🔍 Fuentes RAG utilizadas (${sources.length})</span>
                                <svg class="w-4 h-4 transform group-open:rotate-180 transition" fill="none" stroke="currentColor" viewBox="0 0 24 24"><path stroke-linecap="round" stroke-linejoin="round" stroke-width="2" d="M19 9l-7 7-7-7"></path></svg>
                            </summary>
                            <div class="mt-2 space-y-2 max-h-[160px] overflow-y-auto pr-1">
                                ${sourceItems}
                            </div>
                        </details>
                    </div>
                `;
            }

            bodyDiv.innerHTML = htmlContent;
            chatBox.scrollTop = chatBox.scrollHeight;
        }

        function formatText(text) {
            return text
                .replace(/\n/g, '<br>')
                .replace(/\*\*(.*?)\*\*/g, '<strong>$1</strong>');
        }

        if ('webkitSpeechRecognition' in window || 'SpeechRecognition' in window) {
            const SpeechRecognition = window.SpeechRecognition || window.webkitSpeechRecognition;
            recognition = new SpeechRecognition();
            recognition.continuous = false;
            recognition.lang = 'es-ES';
            recognition.interimResults = false;

            recognition.onstart = () => {
                isRecording = true;
                micBtn.classList.replace('bg-gray-100', 'bg-red-500');
                micBtn.classList.replace('text-gray-600', 'text-white');
                micIcon.classList.add('animate-pulse');
                chatInput.placeholder = "Escuchando... Habla ahora.";
            };

            recognition.onend = () => {
                isRecording = false;
                micBtn.classList.replace('bg-red-500', 'bg-gray-100');
                micBtn.classList.replace('text-white', 'text-gray-600');
                micIcon.classList.remove('animate-pulse');
                chatInput.placeholder = "Escribe tu pregunta sobre la base de conocimientos...";
            };

            recognition.onresult = (event) => {
                chatInput.value = event.results[0][0].transcript;
            };

            micBtn.addEventListener('click', () => {
                if (isRecording) recognition.stop();
                else recognition.start();
            });
        } else {
            micBtn.classList.add('opacity-40', 'cursor-not-allowed');
            micBtn.disabled = true;
        }

        checkConnection();
        setInterval(checkConnection, 10000);
